─────────────────────────────────────────────────────────────────────
import axios from 'axios';

// 1. CREATE THE AXIOS INSTANCE
//
// Think of this as creating YOUR OWN version of axios,
// pre-configured so you don't repeat settings everywhere.
//
const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL, // e.g. "http://localhost:5000/api"
  withCredentials: true,                  // send the refresh cookie automatically
});

// 2. OUTGOING INTERCEPTOR (Request)
//
// This runs BEFORE every request leaves your app.
// Job: "Put the access token in the Authorization header."
//
api.interceptors.request.use((config) => {
  // Get the access token from wherever you store it
  // (e.g., Zustand store, localStorage, memory variable)
  const token = getAccessToken(); // you will fill this in from your auth store

  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }

  return config; // continue sending the request
});

// 3. ONE-AT-A-TIME REFRESH LOGIC
//
// Problem: if 5 requests all fail at once (token expired),
//          we don't want 5 refresh calls - just ONE.
//
// Solution: use a "lock" variable.
//
let isRefreshing = false;          // is a refresh already in progress?
let waitingQueue: Array<{          // requests waiting for the refresh to finish
  resolve: (token: string) => void;
  reject: (error: unknown) => void;
}> = [];

// Helper: resolve or reject all waiting requests
function processQueue(error: unknown, token: string | null) {
  waitingQueue.forEach((prom) => {
    if (error) {
      prom.reject(error);
    } else {
      prom.resolve(token!);
    }
  });
  waitingQueue = [];
}

// 4. INCOMING INTERCEPTOR (Response)
//
// This runs AFTER every response arrives.
// Job: "If the server said 401 (token expired), refresh and retry."
//
api.interceptors.response.use(
  (response) => response, // if response is OK, just pass it through

  async (error) => {
    const originalRequest = error.config;

    // Only handle 401 errors that we haven't already retried
    const isExpired =
      error.response?.status === 401 &&
      !originalRequest._retry; // _retry is our custom flag to avoid infinite loop

    if (!isExpired) {
      return Promise.reject(error); // some other error, don't handle it here
    }

    // Mark this request so we don't retry it again if refresh also fails
    originalRequest._retry = true;

    // Case A: a refresh is ALREADY in progress
    if (isRefreshing) {
      // Wait in line, and retry when refresh is done
      return new Promise((resolve, reject) => {
        waitingQueue.push({
          resolve: (newToken) => {
            originalRequest.headers.Authorization = `Bearer ${newToken}`;
            resolve(api(originalRequest));
          },
          reject,
        });
      });
    }

    // Case B: WE are the first to need a refresh
    isRefreshing = true;

    try {
      // Call the refresh endpoint - the cookie is sent automatically
      const { data } = await axios.post(
        `${import.meta.env.VITE_API_URL}/auth/refresh`,
        {},
        { withCredentials: true }
      );

      const newToken = data.accessToken;

      // Save the new access token (replace with your actual store setter)
      setAccessToken(newToken); // you will fill this in from your auth store

      // Tell all waiting requests: "here is the new token, retry now"
      processQueue(null, newToken);

      // Retry the original request with the new token
      originalRequest.headers.Authorization = `Bearer ${newToken}`;
      return api(originalRequest);

    } catch (refreshError) {
      // Refresh failed - the refresh token is also expired or invalid
      processQueue(refreshError, null);

      // Log the user out (replace with your actual logout action)
      logoutUser(); // you will fill this in from your auth store

      return Promise.reject(refreshError);

    } finally {
      isRefreshing = false; // unlock so future requests can refresh again
    }
  }
);

export default api;

HOW IT ALL FLOWS - Simple Story
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Normal request:
    App -> [outgoing interceptor adds token] -> Server -> Response OK

  Token expired (1 request):
    App -> Server -> 401 expired
    -> [incoming interceptor catches it]
    -> calls /auth/refresh with cookie
    -> gets new token
    -> retries original request -> Response OK

  Token expired (5 requests at the same time):
    Request 1 -> 401 -> starts refresh (isRefreshing = true)
    Requests 2-5 -> 401 -> see isRefreshing = true -> join the queue
    Refresh finishes -> new token distributed to all 5 -> all retry OK

  Refresh token also expired:
    -> processQueue(error) -> all fail
    -> logoutUser() is called -> user is sent to login page

import { QueryClient } from '@tanstack/react-query';

const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      // Don't refetch every time you switch browser tabs.
      // For a chat app, this would cause unnecessary noise.
      refetchOnWindowFocus: false,

      // If a request fails, try ONE more time before showing an error.
      retry: 1,

      // Consider data "fresh" for 1 minute.
      // During this time, React Query will NOT re-fetch it.
      // After 1 min, if you visit the page again, it re-fetches in background.
      staleTime: 1000 * 60, // 60 seconds
    },
  },
});

export default queryClient;

// QUERY KEY NAMING SCHEME
//
// Query keys are labels React Query uses to identify cached data.
// Think of them like folder names:
//   ["conversations"]          -> the full list of conversations
//   ["conversation", "abc123"] -> one specific conversation
//   ["messages", "abc123"]     -> messages inside conversation "abc123"
//
// Rules we follow:
//   1. Always start with a general category string (e.g. "conversations")
//   2. Add an ID for specific items (e.g. "abc123")
//   3. Keep them in ONE place (this file) so there are no typos

export const queryKeys = {
  // All conversations for the logged-in user
  conversations: () => ['conversations'] as const,

  // One specific conversation by its ID
  conversation: (id: string) => ['conversation', id] as const,

  // All messages inside a specific conversation
  messages: (conversationId: string) => ['messages', conversationId] as const,
};

// HOW TO USE:
//
// In a query:
//   useQuery({ queryKey: queryKeys.conversations(), queryFn: ... })
//
// To invalidate (force re-fetch) after sending a message:
//   queryClient.invalidateQueries({ queryKey: queryKeys.messages(convId) })

HOW REACT QUERY WORKS - Simple Story
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  WITHOUT React Query:
    Component -> useEffect -> fetch -> setState(data)
    You manually manage: loading, error, caching, re-fetching...

  WITH React Query:
    Component -> useQuery({ queryKey: ..., queryFn: ... })
    React Query gives you back: { data, isLoading, error }
    AND it caches, re-fetches, deduplicates automatically!

  Real example:
    // In a ConversationList component:
    const { data: conversations, isLoading } = useQuery({
      queryKey: queryKeys.conversations(),     // ["conversations"]
      queryFn: () => api.get('/conversations'),
    });

    // In a MessageList component for conversation "abc123":
    const { data: messages } = useQuery({
      queryKey: queryKeys.messages('abc123'),  // ["messages", "abc123"]
      queryFn: () => api.get('/conversations/abc123/messages'),
    });
